@echo off
title Envío de comprobantes a SUNAT

:: ---------- 1. Iniciar el servicio (sin esperar) ----------
echo Iniciando servicio SFS...
start "" "EjecutarSFS.bat"

:: ---------- 2. Esperar 40 s fijos antes de lanzar nuestro exe ----------
echo Esperando 40 segundos a que el servicio se estabilice...
timeout /t 40 /nobreak >nul

:: ---------- 3. Ejecutar nuestro programa ----------
:ejecutarExe
echo Ejecutando envio...
enviaSunat.exe

:: ---------- 4. Mensaje y abrir bandeja ----------
echo.
echo ========================================
echo proceso terminado
echo ========================================
timeout /t 3 /nobreak >nul
start http://localhost:9000

:: ---------- 5. Cerrar ventana ----------
exit
